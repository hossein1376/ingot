fn f(a: i64, b: i64, xs: []i64) -> i64 {
    let x = a * b + 1 << 2 & 7;
    let y = a < b && b <= a || a == b;
    let z = -a + (b as i64) - -1;
    let p = &mut xs[0];
    let q = p.*;
    let r = xs[..];
    let s = xs[..2];
    let t = xs[2..];
    let u = xs[1..3];
    let v = f(a, b, xs);
    let w = xs[0].cmp(&xs[1]);
    return x + y + z + q.* + r.len + s.len + t.len + u.len + v + w;
}

fn variants() -> Option[i64] {
    let a = .some(1);
    let b = .none;
    let c = Option.some(2);
    return if true { a } else { b };
}

fn literals() -> i64 {
    let a = [3]i64.{ 1, 2, 3 };
    let b = Point.{ x: 1, y: 2 };
    let c = .{ x: 3, y: 4 };
    let d = (1, 2, 3);
    let e = fn [&mut a, b](x: i64) -> i64 { return x + a; };
    let f = "text\n";
    let g = c"bytes";
    let h = 'x';
    return 1.5 as i64 + 2e3 as i64 + 0xff;
}

fn fallible(x: i64) -> i64 ! Error {
    let a = try might(x);
    let b = might(x) catch _ => 0;
    let c = maybe(x) orelse 1;
    return a + b + c;
}

fn control(c: bool, x: i64) -> i64 {
    let a = if c { 1 } else { return 2; };
    let b = match x {
        .some(v) => v,
        .none => { return 0; }
    };
    return a + b;
}

fn unsafe_spawn(x: i64) -> i64 {
    let p: *i64 = @uninit(*i64);
    return unsafe("p is live") { p.* };
}
