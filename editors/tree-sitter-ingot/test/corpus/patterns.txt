copy struct Pair { a: i64, b: i64 }

fn rest(p: Pair) -> i64 {
    let Pair.{ a, .. } = p;
    let .{ b, .. } = p;
    let Pair.{ } = p;
    return a + b;
}

fn arms(x: ?i64, p: Pair) -> i64 {
    return match x {
        .some(1..=9) => 1,
        .some(v) if v > 20 => v,
        .some(_) => 2,
        .none | else => 0,
    };
}

fn fields(p: Pair) -> i64 {
    return match p {
        Pair.{ a: 0, .. } => 0,
        Pair.{ a, b: 1 } => a,
        Pair.{ .. } => 1,
    };
}

fn tuples(x: (i64, bool)) -> i64 {
    return match x {
        (0, true) => 1,
        (a, false) => a,
        _ => 3,
    };
}
